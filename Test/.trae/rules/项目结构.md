项目结构
studio-safety-system/
├── server/                          # Node.js + Express 后端
│   ├── app.js                       # 入口（含 node-cron 定时任务）
│   ├── init.sql                     # 数据库建表 + 预置数据
│   ├── config/db.js                 # MySQL 连接池
│   ├── middleware/
│   │   ├── auth.js                  # JWT 认证 + 角色校验中间件
│   │   └── upload.js               # Multer 图片上传
│   ├── routes/
│   │   ├── auth.js                  # 登录、获取用户信息
│   │   ├── studios.js              # 工作室 CRUD
│   │   ├── inspections.js          # 检查提交/审核/整改/确认
│   │   ├── notifications.js        # 通知列表/已读/提醒
│   │   ├── statistics.js           # 未检查名单/问题分析/排名/报表/Excel导出
│   │   └── settings.js             # 系统设置/发布通知
│   └── utils/
│       ├── watermark.js            # Sharp 图片水印
│       └── notification.js         # 消息创建/定时检查逻辑
├── client/                          # Vue 3 + Ant Design Vue 前端
│   ├── src/
│   │   ├── api/                     # 6 个 axios API 模块
│   │   ├── components/              # AppLayout, PhotoUploader, PhotoViewer, NotificationBell
│   │   ├── views/                   # 10 个页面（20 个 .vue 文件）
│   │   ├── store/                   # Pinia stores (auth, notification)
│   │   ├── router/index.js          # Vue Router + 权限路由守卫
│   │   └── utils/constants.js       # 状态/角色常量映射
│   └── vite.config.js              # Vite + 代理配置
